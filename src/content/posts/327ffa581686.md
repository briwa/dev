const BRANCH_WIDTH = 2;
const START_X = (width / 2) - BRANCH_WIDTH / 2;
const START_Y = height;

function createBranch(props) {
  return new Canvas.Line({
    ...props,
    color: { r: 224, g: 122, b: 95 },
  });
}

const entities = [];
const steps = [new Canvas.Step({ duration: 10 })];

// first branch is gonna be at least 30% the canvas height
// it'll then split to two, 30% of its previous distance, angled +- 10 deg of its original vector
// let's say, the maximum branching is 2^8

let count = 0;
let startX = START_X;
let startY = START_Y;
let xOffset = 0;
let yOffset = 0;
let yOffsetRatio = 1;
const MAX_STEP = 3;
const TARGET_COUNT = Math.pow(2, 8);
const X_DIFF = 10;
const Y_DIFF_RATIO = [0.3, 0.1]; // of height

function negateOnOdd(val, num) {
  return val * (num % 2 === 0 ? 1 : -1);
}

function exp(start, end, t) {
  return start * Math.pow(end / start, t);
}

for (let step = 0; count < Math.pow(2, MAX_STEP); step++) {
  const t = step / MAX_STEP;
  count = Math.pow(2, step);
  halfCount = Math.max(Math.floor(count / 2), 0.5);
  xOffset = X_DIFF * step;
  yOffset = height * exp(Y_DIFF_RATIO[0], Y_DIFF_RATIO[1], t);
  
  for (let i = 0; i < halfCount * 2; i++) {
    const x0 = startX;
    const y0 = startY;
    const x1 = startX + negateOnOdd(xOffset, i);
    const y1 = startY - yOffset;
    entities.push(createBranch({
      x0,
      y0,
      x1,
      y1,
    }));
  }

  startY = startY - yOffset;
}

const timeline = new Canvas.Timeline(entities, steps);

function render(timeline) {
  const renderer = new Canvas.Renderer(canvas, timeline.entities);
  
  loop((t) => {
    timeline.update(t);
    renderer.update(t);
  
    if (timeline.done) {
      reset();
      timeline.reset();
    }
  });
}

render(timeline);
