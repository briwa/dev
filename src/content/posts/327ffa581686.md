Back in uni, there was this assignment. We were supposed to present some topic in front of the class, with PowerPoint and all that. We would get some sort of 1st place awards based on a few aspects, including the presentation itself. I wasn't very much into the assignment at all, however, I enjoyed making slides. So, I thought I'd get the "best visual award" or something, by making it look fancy with transitions and all that.

Then came my friend, presenting it on Adobe _freaking_ Flash Player. Yep, I lost, right then and there lol. The graphics were on point, the animations were so smooth, and coupled with the very sophisticated topic he came up with, he would eventually got the 1st place.

What was stuck with me was this animation that he did; it was a simple vines growing. I didn't ask him how he did it, out of jealousy. Looking back, I'd like to replicate that. It is a very interesting concept to me for a while: fractals. [Wikipedia says](https://en.wikipedia.org/wiki/Fractal) it is a form of recursive algorithm or some sort.

---

The concept is in fact pretty simple. Starting out with a single branch, growing from one point to another. At the end of it, it will grow into a number of similar branches, perhaps shorter, separated from each other by some angles. The same rules, repeated over and over, and in the end, it'll resemble something of what my friend did a while back.

This time, I'll recreate it in the simplest manner: a "tree" consists of a group of branches. For each branch, it'll grow into 3 branches, and so on and so forth. Here's the visualization:

```sandbox=js viz 480x480
class Tree {
  constructor({ firstBranch, maxDepth, branchCount }) {
    this.maxDepth = maxDepth;
    this.branchCount = branchCount;
    this.branches = [];

    new Branch({ tree: this, attrs: firstBranch, depth: 0 });
  }

  assign(branch) {
    (this.branches[branch.depth] ??= []).push(branch);
  }
}

class Branch {
  constructor({ tree, attrs, depth }) {
    this.depth = depth;
    this.attrs = attrs;
    this.tree = tree;
    this.tree.assign(this);

    if (this.depth <= this.tree.maxDepth) {
      const angleDiff = ANGLE_DIFF / this.tree.branchCount;
      const center = (this.tree.branchCount - 1) / 2;
      const lineWidth = this.attrs.lineWidth - 1.5;

      for (let i = 0; i < this.tree.branchCount; i++) {
        const offset = i - center;
        const angle = this.attrs.angle + (offset * angleDiff);
        const length = this.attrs.length * DISTANCE_RATIO;

        new Branch({
          tree: this.tree,
          attrs: {
            start: this.end,
            length,
            angle,
            lineWidth,
          },
          depth: this.depth + 1,
        });
      }
    }
  }

  get end() {
    const { start, angle, length } = this.attrs;
    const angleInRad = angle / 180 * Math.PI;

    return {
      x: start.x - (Math.sin(angleInRad) * length),
      y: start.y - (Math.cos(angleInRad) * length),
    }
  }
}

const tree = new Tree({
  firstBranch: {
    start: { x: START_X, y: START_Y },
    lineWidth: BASE_THICKNESS,
    length: BASE_LENGTH,
    angle: 0,
  },
  maxDepth: 6,
  branchCount: 3,
});

render(tree);

```

We could add some more randomizing factors in the angles, the length, but for now, I kept it simple. I guess it isn't so bad for a 2nd place :p

```sandbox=js label="shared source"
const DISTANCE_RATIO = 0.7;
const BASE_LENGTH = 110;
const BASE_THICKNESS = 8;
const ANGLE_DIFF = 100;
const START_X = (width / 2) - (BASE_THICKNESS / 2);
const START_Y = height;
const GROW_STEP_DURATION = 300;

function toCurve(branch) {
  return Canvas.curve({
    x0: branch.attrs.start.x,
    y0: branch.attrs.start.y,
    x1: branch.end.x,
    y1: branch.end.y,
    lineWidth: branch.attrs.lineWidth,
    ease: Canvas.easeInOutSine,
    color: { r: 224, g: 122, b: 95 },
  });
}

function render(tree) {
  const entitiesByBranches = tree.branches.map((branches) => branches.map(toCurve));
  const entities = entitiesByBranches.flat();

  class GrowStep extends Canvas.TweenerStep {
      enter() {
        let startAt = 0;
    
        entitiesByBranches.forEach((entitiesInBranches) => {
          entitiesInBranches.forEach((entity) => {
            this.tween(entity, {
              startAt,
              duration: GROW_STEP_DURATION,
              ease: Canvas.easeInOutSineInverse,
              to: { t1: 1 },
            });
          });
    
          startAt += GROW_STEP_DURATION;
        });
      }
    }
  
  class FadeOutStep extends Canvas.TweenerStep {
    enter() {
      this.entities.forEach((entity) => {
        this.tween(entity, {
          startAt: 0,
          duration: this.duration,
          to: { alpha: 0 },
        });
      });
    }
  }
  
  class ResetStep extends Canvas.DurationStep {
    enter() {
      this.entities.forEach((entity) => {
        entity.t1 = 0;
        entity.alpha = 1;
      });
    }
  }

  const steps = [
    new Canvas.DurationStep({ duration: 10 }),
    new FadeOutStep({ duration: 800 }),
    new ResetStep({ duration: 10 }),
    new GrowStep({
      duration: entitiesByBranches.length * GROW_STEP_DURATION,
    }),
  ];
  
  const timeline = new Canvas.Timeline({ entities, steps });
  const scene = new Canvas.Scene({
    canvas,
    timelines: [timeline],
  });
  
  loop((t) => {
    scene.render(t);
    if (timeline.finished) reset();
  });
}
```
