const BRANCH_WIDTH = 2;
const START_X = (width / 2) - BRANCH_WIDTH / 2;
const START_Y = height;

const entities = [];

function createBranch(props) {
  entities.push(new Canvas.Line({
    ...props,
    color: { r: 224, g: 122, b: 95 },
  }));
}

const steps = [new Canvas.Step({ duration: 10 })];

function negateOnOdd(val, num) {
  return val * (num % 2 === 0 ? 1 : -1);
}

function exp(start, end, t) {
  return start * Math.pow(end / start, t);
}

// For every branch, there must be two branches produced
// height is 20% smaller than its parent
// each is angled +/- 20 * x/2 degrees from its parent
// example: X branch: 90 deg, 100px
// - A branch: 80 deg, 80px
//   - AA branch: 75 deg, 60px
//   - AB branch: 85 deg, 60px
// - B branch: 100 deg, 80px
//   - BA branch: 105 deg, 60px
//   - BB branch: 95 deg, 60px

let count = 0;
let startX = START_X;
let startY = START_Y;
let xOffset = 0;
let yOffset = 0;
let yOffsetRatio = 1;
const MAX_STEP = 3;
const X_DIFF = 10;
const Y_DIFF_RATIO = [0.2, 0.05]; // of height

class Branch {
  constructor(props) {
    this.props = props;
    createBranch(props);

    if (entities.length < Math.pow(2, MAX_STEP)) {
      new Branch(this.left);
      new Branch(this.right); 
    }
  }

  get height() {
    return this.props.y0 - this.props.y1;
  }

  get newY1() {
    return this.props.y1 - (this.height * 0.8);
  }

  get left() {
    return {
      x0: this.props.x1,
      y0: this.props.y1,
      x1: this.props.x1 - X_DIFF,
      y1: this.newY1,
    }
  }

  get right() {
    return {
      x0: this.props.x1,
      y0: this.props.y1,
      x1: this.props.x1 + X_DIFF,
      y1: this.newY1,
    }
  }
}

new Branch({
  x0: START_X,
  x1: START_X,
  y0: START_Y,
  y1: START_Y - 100,
});

const timeline = new Canvas.Timeline(entities, steps);

function render(timeline) {
  const renderer = new Canvas.Renderer(canvas, timeline.entities);
  
  loop((t) => {
    timeline.update(t);
    renderer.update(t);
  
    if (timeline.done) {
      reset();
      timeline.reset();
    }
  });
}

render(timeline);
