const DISTANCE_RATIO = 0.7;
const BASE_LENGTH = 110;
const BASE_THICKNESS = 8;
const ANGLE_DIFF = 100;
const START_X = (width / 2) - (BASE_THICKNESS / 2);
const START_Y = height;
const generations = {};

class Tree {
  constructor({ firstBranch, maxGenerations, branchCount }) {
    this.maxGenerations = maxGenerations;
    this.branchCount = branchCount;
    this.generations = {};

    new Branch({ tree: this, attrs: firstBranch, generation: 0 });
  }

  assign(branch) {
    (this.generations[branch.generation] ??= []).push(branch);
  }
}

class Branch {
  constructor({ tree, attrs, generation }) {
    this.generation = generation;
    this.attrs = attrs;
    this.tree = tree;
    this.tree.assign(this);

    if (this.generation <= this.tree.maxGenerations) {
      const angleDiff = ANGLE_DIFF / this.tree.branchCount;
      const center = (this.tree.branchCount - 1) / 2;
      const lineWidth = this.attrs.lineWidth - 1.5;

      for (let i = 0; i < this.tree.branchCount; i++) {
        const offset = i - center;
        const angle = this.attrs.angle + (offset * angleDiff);
        const length = this.attrs.length * DISTANCE_RATIO;

        new Branch({
          tree: this.tree,
          attrs: {
            start: this.end,
            length,
            angle,
            lineWidth,
          },
          generation: this.generation + 1,
        });
      }
    }
  }

  get end() {
    const { start, angle, length } = this.attrs;
    const angleInRad = angle / 180 * Math.PI;

    return {
      x: start.x - (Math.sin(angleInRad) * length),
      y: start.y - (Math.cos(angleInRad) * length),
    }
  }
}

const tree = new Tree({
  firstBranch: {
    start: { x: START_X, y: START_Y },
    lineWidth: BASE_THICKNESS,
    length: BASE_LENGTH,
    angle: 0,
  },
  maxGenerations: 6,
  branchCount: 3,
});

function toCurve(branch) {
  return Canvas.curve({
    x0: branch.attrs.start.x,
    y0: branch.attrs.start.y,
    x1: branch.end.x,
    y1: branch.end.y,
    lineWidth: branch.attrs.lineWidth,
    ease: Canvas.easeInOutSine,
    color: { r: 224, g: 122, b: 95 },
  });
}

const entitiesByGeneration = Object.values(tree.generations)
  .map((branches) => branches.map(toCurve));
const entities = entitiesByGeneration.flat();

const GROW_STEP_DURATION = 300;
class GrowStep extends Canvas.TweenerStep {
  enter() {
    let startAt = 0;

    entitiesByGeneration.forEach((entitiesInGeneration) => {
      entitiesInGeneration.forEach((entity) => {
        this.tween(entity, {
          startAt,
          duration: GROW_STEP_DURATION,
          ease: Canvas.easeInOutSineInverse,
          to: { t1: 1 },
        });
      });

      startAt += GROW_STEP_DURATION;
    });
  }
}

class FadeOutStep extends Canvas.TweenerStep {
  enter() {
    this.entities.forEach((entity) => {
      this.tween(entity, {
        startAt: 0,
        duration: this.duration,
        to: { alpha: 0 },
      });
    });
  }
}

class ResetStep extends Canvas.DurationStep {
  enter() {
    this.entities.forEach((entity) => {
      entity.t1 = 0;
      entity.alpha = 1;
    });
  }
}

const steps = [
  new Canvas.DurationStep({ duration: 10 }),
  new FadeOutStep({ duration: 800 }),
  new ResetStep({ duration: 10 }),
  new GrowStep({
    duration: tree.maxGenerations * GROW_STEP_DURATION,
  }),
];

const timeline = new Canvas.Timeline({ entities, steps });
const scene = new Canvas.Scene({
  canvas,
  timelines: [timeline],
});

loop((t) => {
  scene.render(t);
  if (timeline.finished) reset();
});
