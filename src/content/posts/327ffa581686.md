const BRANCH_WIDTH = 2;
const START_X = (width / 2) - BRANCH_WIDTH / 2;
const START_Y = height;
const entities = [];
const steps = [new Canvas.Step({ duration: 10 })];

function exp(start, end, t) {
  return start * Math.pow(end / start, t);
}

function moveBy({ start, distance, angle }) {
  const angleInRad = angle / 180 * Math.PI;

  return {
    x: start.x + (Math.sin(angleInRad) * distance),
    y: start.y + (Math.cos(angleInRad) * distance),
  }
}

const MAX_STEP = 8;

class Branch {
  constructor(options, step = 0) {
    this.props = this.getProps(options);
    entities.push(new Canvas.Line({
      ...this.props,
      color: { r: 224, g: 122, b: 95 },
    }));

    if (step < MAX_STEP) {
      const newOptions = {
        start: { x: this.props.x1, y: this.props.y1 },
        distance: options.distance * 0.7,
      };

      new Branch({
        ...newOptions,
        angle: options.angle - 25,
      }, step + 1);

      new Branch({
        ...newOptions,
        angle: options.angle + 25,
      }, step + 1);
    }
  }

  getProps (options) {
    const target = moveBy(options);

    return {
      x0: options.start.x,
      y0: options.start.y,
      x1: target.x,
      y1: target.y,
    };
  }
}

new Branch({
  start: { x: START_X, y: START_Y },
  distance: 100,
  angle: -180,
});

const timeline = new Canvas.Timeline(entities, steps);
function render(timeline) {
  const renderer = new Canvas.Renderer(canvas, timeline.entities);
  
  loop((t) => {
    timeline.update(t);
    renderer.update(t);
  
    if (timeline.done) {
      reset();
      timeline.reset();
    }
  });
}

render(timeline);

const BRANCH_WIDTH = 2;
const START_X = (width / 2) - BRANCH_WIDTH / 2;
const START_Y = height;

const entities = [];

function createBranch(props) {
  entities.push(new Canvas.Line({
    ...props,
    color: { r: 224, g: 122, b: 95 },
  }));
}

const steps = [new Canvas.Step({ duration: 10 })];

function exp(start, end, t) {
  return start * Math.pow(end / start, t);
}

function moveBy(distance, angle) {
  const angleInRad = angle / 180 * Math.PI;
  return {
    x: Math.sin(angleInRad) * distance,
    y: Math.cos(angleInRad) * distance,
  }
}

const MAX_STEP = 8;
const Y_DIFF_RATIO = [0.2, 0.05]; // of height

class Branch {
  constructor(props, step = 0) {
    this.props = props;
    createBranch(props);

    if (step < MAX_STEP) {
      new Branch(this.left, step + 1);
      new Branch(this.right, step + 1); 
    }
  }

  get height() {
    return this.props.y0 - this.props.y1;
  }

  get newY1() {
    return this.props.y1 - (this.height * 0.6);
  }

  get xDiff() {
    return this.height * 0.9;
  }

  get left() {
    return {
      x0: this.props.x1,
      y0: this.props.y1,
      x1: this.props.x1 - this.xDiff,
      y1: this.newY1,
    }
  }

  get right() {
    return {
      x0: this.props.x1,
      y0: this.props.y1,
      x1: this.props.x1 + this.xDiff,
      y1: this.newY1,
    }
  }
}

new Branch({
  x0: START_X,
  x1: START_X,
  y0: START_Y,
  y1: START_Y - 100,
});

const timeline = new Canvas.Timeline(entities, steps);
function render(timeline) {
  const renderer = new Canvas.Renderer(canvas, timeline.entities);
  
  loop((t) => {
    timeline.update(t);
    renderer.update(t);
  
    if (timeline.done) {
      reset();
      timeline.reset();
    }
  });
}

render(timeline);
