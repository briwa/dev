const MAX_GENERATIONS = 7;
const DISTANCE_RATIO = 0.7;
const BASE_LENGTH = 110;
const BASE_THICKNESS = 8;
const ANGLE_DIFF = 100;
const START_X = (width / 2) - (BASE_THICKNESS / 2);
const START_Y = height;
const generations = {};

function moveBy({ start, length, angle }) {
  const angleInRad = angle / 180 * Math.PI;

  return {
    x: start.x - (Math.sin(angleInRad) * length),
    y: start.y - (Math.cos(angleInRad) * length),
  }
}

class Branch {
  constructor(options, generation = 0) {
    this.generation = generation;
    const curve = this.assignCurve(options);

    if (generation < MAX_GENERATIONS) {
      const newBranchCount = 3;
      const angleDiff = ANGLE_DIFF / newBranchCount;
      const nextGeneration = generation + 1;
      const center = (newBranchCount - 1) / 2;
      const lineWidth = options.lineWidth - 1.5;

      for (let i = 0; i < newBranchCount; i++) {
        const offset = i - center;
        const angle = options.angle + (offset * angleDiff);
        const length = options.length * DISTANCE_RATIO;

        new Branch({
          start: { x: curve.x1, y: curve.y1 },
          length,
          angle,
          lineWidth,
        }, nextGeneration);
      }
    }
  }

  assignCurve(options) {
    const target = moveBy(options);
    const curve = Canvas.curve({
      x0: options.start.x,
      y0: options.start.y,
      x1: target.x,
      y1: target.y,
      ease: Canvas.easeInOutSine,
      lineWidth: options.lineWidth,
      color: { r: 224, g: 122, b: 95 },
    });

    curve.generation = this.generation;
    (generations[this.generation] ??= []).push(curve);
    
    return curve;
  }
}

new Branch({
  start: { x: START_X, y: START_Y },
  lineWidth: BASE_THICKNESS,
  length: BASE_LENGTH,
  angle: 0,
});

const entities = Object.values(generations).flat();

const GROW_STEP_DURATION = 300;
class GrowStep extends Canvas.TweenerStep {
  enter() {
    let startAt = 0;
    for (let i = 0; i < MAX_GENERATIONS; i++) {
      generations[i].forEach((entity) => {
        this.tween(entity, {
          startAt,
          duration: GROW_STEP_DURATION,
          ease: Canvas.easeInOutSineInverse,
          to: { t1: 1 },
        });
      });

      startAt += GROW_STEP_DURATION;
    }
  }
}

class FadeOutStep extends Canvas.TweenerStep {
  enter() {
    this.entities.forEach((entity) => {
      this.tween(entity, {
        startAt: 0,
        duration: this.duration,
        to: { alpha: 0 },
      });
    });
  }
}

class ResetStep extends Canvas.DurationStep {
  enter() {
    this.entities.forEach((entity) => {
      entity.t1 = 0;
      entity.alpha = 1;
    });
  }
}

const steps = [
  new Canvas.DurationStep({ duration: 10 }),
  new FadeOutStep({ duration: 800 }),
  new ResetStep({ duration: 10 }),
  new GrowStep({
    duration: MAX_GENERATIONS * GROW_STEP_DURATION,
  }),
];

const timeline = new Canvas.Timeline({ entities, steps });
const scene = new Canvas.Scene({
  canvas,
  timelines: [timeline],
});

loop((t) => {
  scene.render(t);
  if (timeline.finished) reset();
});
