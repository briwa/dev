const MAX_STEP = 7;
const DISTANCE_RATIO = 0.7;
const BASE_LENGTH = 110;
const BASE_THICKNESS = 8;
const ANGLE_DIFF = 100;
const START_X = (width / 2) - (BASE_THICKNESS / 2);
const START_Y = height;
const entities = [];
const steps = [new Canvas.BaseStep()];

function moveBy({ start, length, angle }) {
  const angleInRad = angle / 180 * Math.PI;

  return {
    x: start.x - (Math.sin(angleInRad) * length),
    y: start.y - (Math.cos(angleInRad) * length),
  }
}

class Branch {
  constructor(options, step = 0) {
    const nextStart = this.draw(options);

    if (step < MAX_STEP) {
      const nextCount = 3;
      const angleDiff = ANGLE_DIFF / nextCount;
      const nextStep = step + 1;
      const newOptions = {
        start: nextStart,
        length: options.length * DISTANCE_RATIO,
      };

      const center = (nextCount - 1) / 2;
      const lineWidth = options.lineWidth - 1.5;
      for (let i = 0; i < nextCount; i++) {
        const offset = i - center;
        const angle = options.angle - (offset * angleDiff);

        new Branch({ ...newOptions, angle, lineWidth }, nextStep);
      }
    }
  }

  draw(options) {
    const target = moveBy(options);
    const props = {
      x0: options.start.x,
      y0: options.start.y,
      x1: target.x,
      y1: target.y,
      lineWidth: options.lineWidth,
      color: { r: 224, g: 122, b: 95 },
    };

    entities.push(Canvas.line(props));
    
    return { x: props.x1, y: props.y1 };
  }
}

new Branch({
  start: { x: START_X, y: START_Y },
  lineWidth: BASE_THICKNESS,
  length: BASE_LENGTH,
  angle: 0,
});

const scene = new Canvas.Scene({
  canvas,
  loop: true,
  timelines: [new Canvas.Timeline({ entities, steps })],
});

loop((t) => scene.render(t));

const MAX_STEP = 7;
const DISTANCE_RATIO = 0.7;
const BASE_LENGTH = 110;
const BASE_THICKNESS = 8;
const ANGLE_DIFF = 100;
const START_X = (width / 2) - (BASE_THICKNESS / 2);
const START_Y = height;
const entities = [];

function moveBy({ start, length, angle }) {
  const angleInRad = angle / 180 * Math.PI;

  return {
    x: start.x - (Math.sin(angleInRad) * length),
    y: start.y - (Math.cos(angleInRad) * length),
  }
}

class Branch {
  constructor(options, step = 0) {
    this.step = step;
    const nextStart = this.draw(options);

    if (step < MAX_STEP) {
      const nextCount = 3;
      const angleDiff = ANGLE_DIFF / nextCount;
      const nextStep = step + 1;
      const newOptions = {
        start: nextStart,
        length: options.length * DISTANCE_RATIO,
      };

      const center = (nextCount - 1) / 2;
      const lineWidth = options.lineWidth - 1.5;
      for (let i = 0; i < nextCount; i++) {
        const offset = i - center;
        const angle = options.angle - (offset * angleDiff);

        new Branch({ ...newOptions, angle, lineWidth }, nextStep);
      }
    }
  }

  draw(options) {
    const target = moveBy(options);
    const curve = Canvas.curve({
      x0: options.start.x,
      y0: options.start.y,
      x1: target.x,
      y1: target.y,
      t0: 0,
      t1: 0,
      ease: Canvas.easeInOutSine,
      lineWidth: options.lineWidth,
      color: { r: 224, g: 122, b: 95 },
    });

    curve.step = this.step;
    entities.push(curve);
    
    return { x: curve.x1, y: curve.y1 };
  }
}

new Branch({
  start: { x: START_X, y: START_Y },
  lineWidth: BASE_THICKNESS,
  length: BASE_LENGTH,
  angle: 0,
});

const DURATION_PER_BRANCH = 300;

class GrowStep extends Canvas.TweenerStep {
  constructor({ duration, entities }) {
    super({ duration, entities });
  }

  enter() {
    let startAt = 0;
    
    entities.forEach((entity) => {
      this.tween(entity, {
        startAt: 0,
        duration: DURATION_PER_BRANCH,
        to: { t1: 1 },
      });
    });
  }
}

const steps = [
  new GrowStep({
    duration: entities.length * DURATION_PER_BRANCH,
    entities,
  }),
];

const scene = new Canvas.Scene({
  canvas,
  loop: true,
  timelines: [new Canvas.Timeline({ entities, steps })],
});

loop((t) => scene.render(t));
