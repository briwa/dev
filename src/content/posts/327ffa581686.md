const BRANCH_WIDTH = 2;
const START_X = (width / 2) - BRANCH_WIDTH / 2;
const START_Y = height;
const entities = [];
const steps = [new Canvas.DurationStep({ duration: 10 })];

function moveBy({ start, distance, angle }) {
  const angleInRad = angle / 180 * Math.PI;

  return {
    x: start.x + (Math.sin(angleInRad) * distance),
    y: start.y + (Math.cos(angleInRad) * distance),
  }
}

const MAX_STEP = 8;

class Branch {
  constructor(options, step = 0) {
    this.props = this.getProps(options);
    this.step = step;
    this.draw();

    if (step < MAX_STEP) {
      const newOptions = {
        start: { x: this.props.x1, y: this.props.y1 },
        distance: options.distance * 0.7,
      };

      new Branch({
        ...newOptions,
        angle: options.angle - 25,
      }, step + 1);

      new Branch({
        ...newOptions,
        angle: options.angle + 25,
      }, step + 1);
    }
  }

  draw() {
    entities.push(Canvas.line({
      ...this.props,
      lineWidth: MAX_STEP - this.step,
      color: { r: 224, g: 122, b: 95 },
    }));
  }

  getProps(options) {
    const target = moveBy(options);

    return {
      x0: options.start.x,
      y0: options.start.y,
      x1: target.x,
      y1: target.y,
    };
  }
}

new Branch({
  start: { x: START_X, y: START_Y },
  distance: 100,
  angle: -180,
});

const scene = new Canvas.Scene({
  canvas,
  loop: true,
  timelines: [new Canvas.Timeline({ entities, steps })],
});

loop((t) => scene.render(t));
