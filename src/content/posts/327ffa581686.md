const MAX_STEP = 7;
const DISTANCE_RATIO = 0.7;
const BASE_DISTANCE = 80;
const BASE_BRANCH_WIDTH = 8;
const ANGLE_DIFF = 80;
const START_X = (width / 2) - (BASE_BRANCH_WIDTH / 2);
const START_Y = height;
const entities = [];
const steps = [new Canvas.BaseStep()];

function moveBy({ start, distance, angle }) {
  const angleInRad = angle / 180 * Math.PI;

  return {
    x: start.x - (Math.sin(angleInRad) * distance),
    y: start.y - (Math.cos(angleInRad) * distance),
  }
}

class Branch {
  constructor(options, step = 0) {
    const nextStart = this.draw(options);

    if (step < MAX_STEP) {
      const nextCount = 3;
      const angleDiff = ANGLE_DIFF / nextCount;
      const nextStep = step + 1;
      const newOptions = {
        start: nextStart,
        distance: options.distance * DISTANCE_RATIO,
      };

      const center = (nextCount - 1) / 2;
      const lineWidth = options.lineWidth - 1.5;
      for (let i = 0; i < nextCount; i++) {
        const offset = i - center;
        const angle = options.angle - (offset * angleDiff);

        new Branch({ ...newOptions, angle, lineWidth }, nextStep);
      }
    }
  }

  draw(options) {
    const target = moveBy(options);
    const props = {
      x0: options.start.x,
      y0: options.start.y,
      x1: target.x,
      y1: target.y,
      lineWidth: options.lineWidth,
      color: { r: 224, g: 122, b: 95 },
    };

    entities.push(Canvas.line(props));
    
    return { x: props.x1, y: props.y1 };
  }
}

new Branch({
  start: { x: START_X, y: START_Y },
  lineWidth: BASE_BRANCH_WIDTH,
  distance: BASE_DISTANCE,
  angle: 0,
});

const scene = new Canvas.Scene({
  canvas,
  loop: true,
  timelines: [new Canvas.Timeline({ entities, steps })],
});

loop((t) => scene.render(t));

const MAX_STEP = 8;
const ANGLE_DIFF = 90;
const MIN_DISTANCE_RATIO = 0.5;
const DISTANCE_RATIO_OFFSET = 0.2;
const START_BRANCH_WIDTH = MAX_STEP;
const START_X = (width / 2) - (START_BRANCH_WIDTH / 2);
const START_Y = height;
const entities = [];
const steps = [new Canvas.BaseStep()];

function moveBy({ start, distance, angle }) {
  const angleInRad = angle / 180 * Math.PI;

  return {
    x: start.x - (Math.sin(angleInRad) * distance),
    y: start.y - (Math.cos(angleInRad) * distance),
  }
}

class Branch {
  constructor(options, step = 0) {
    this.props = this.getProps(options);
    this.step = step;
    this.draw();

    const count = 2 + Math.floor(Math.random() * 3);
    const angleDiff = ANGLE_DIFF / count;
    const nextStep = step + 1;
    const isOdd = count % 2 === 0;

    if (step < MAX_STEP) {
      const newOptions = {
        start: { x: this.props.x1, y: this.props.y1 },
        distance: options.distance * this.distanceRatio,
      };

      const center = (count - 1) / 2;
      for (let i = 0; i < count; i++) {
        const offset = i - center;
        const angle = options.angle - (offset * angleDiff);

        new Branch({ ...newOptions, angle }, nextStep);
      }
    }
  }

  get distanceRatio() {
    return MIN_DISTANCE_RATIO + (Math.random() * DISTANCE_RATIO_OFFSET);
  }

  draw() {
    entities.push(Canvas.line({
      ...this.props,
      color: { r: 224, g: 122, b: 95 },
    }));
  }

  getProps(options) {
    const target = moveBy(options);

    return {
      x0: options.start.x,
      y0: options.start.y,
      x1: target.x,
      y1: target.y,
    };
  }
}

new Branch({
  start: { x: START_X, y: START_Y },
  distance: 100,
  angle: 0,
});

const scene = new Canvas.Scene({
  canvas,
  loop: true,
  timelines: [new Canvas.Timeline({ entities, steps })],
});

loop((t) => scene.render(t));
