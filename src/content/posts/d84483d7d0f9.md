const { Scene, layer, line, easeInOutSine, linear } = Canvas;
const COLOR = { r: 224, g: 122, b: 95 };
const BASE_Y = height / 2;
const MAX_RIPPLE = 5;

function drawRipple(y, rippleCount) {
  return line({
    x0: 0,
    x1: width,
    y0: y,
    y1: y,
    t0: 0,
    t1: 1,
    segments: 100,
    color: COLOR,
    ease: (t) => {
      return Math.sin(t * Math.PI * 2 * rippleCount);
    },
  });
}

function waves() {
  const group = [];

  let y = BASE_Y;
  const distance = height / 2;
  for (let i = 0; i < MAX_RIPPLE; i++) {
    const phase = i / MAX_RIPPLE;
    const y = BASE_Y + easeInOutSine(phase) * distance;
    group.push(drawRipple(y, Math.abs(i - MAX_RIPPLE)));
  }
  
  return layer(group);
};
