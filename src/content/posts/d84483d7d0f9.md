```sandbox=js viz 480x480
// draw a line across the canvas vertically, starting somewhere on the first half of the canvas
// then a few more to the right, making something like ripples
// then it pulls and pushes
const { Scene, layer, line, easeInOutSine, linear } = Canvas;
const COLOR = { r: 224, g: 122, b: 95 };

function drawRipple(y) {
  return line({
    x0: 0,
    x1: width,
    y0: y - 5,
    y1: y + 5,
    t0: 0,
    t1: 1,
    segments: 100,
    color: COLOR,
    ease: (t) => {
      return Math.sin(t * Math.PI * 2 * 5);
    },
  });
}

function waves() {
  const baseY = height / 2;
  const group = [drawRipple(baseY)];

  let y = baseY;
  const maxRipple = 5;
  const riplleDistance = height / 6;
  for (let i = 1; i <= maxRipple; i++) {
    y += riplleDistance * easeInOutSine(i / maxRipple);
    group.push(drawRipple(y));
  }

  // phase out lines
  
  
  return layer(group);
};

const scene = new Scene({
  canvas,
  layers: [waves()],
});

scene.onFinish(() => reset());
loop((t) => scene.render(t));

```
