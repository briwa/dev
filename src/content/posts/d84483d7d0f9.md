```js sandbox=canvas 400x400
const {
  Scene,
  tween,
  layer,
  path,
  easeInSine,
  easeOutSine,
  linear,
  move,
  parallel,
  repeat,
  sequence,
  wait,
} = Canvas;
const COLOR = { r: 224, g: 122, b: 95 };
const BASE_Y = height - 100;
const MAX_RIPPLE = 1;

function drawRipple(y) {
  return path({
    x0: 0,
    x1: width,
    y0: y,
    y1: y,
    color: COLOR,
    segments: 96,
    cycles: 2,
    amplitude: 10,
    phase: 0,
    offset: (t, { amplitude, phase, cycles }) => {
      return amplitude * Math.sin(Math.PI * 2 * t * cycles + phase);
    },
  });
}

function waves() {
  return layer(
    drawRipple(BASE_Y),
    sequence(
      move({ y: -100 }, { duration: 1000, ease: easeInSine }),
      move({ y: -50 }, { duration: 5000, ease: easeOutSine }),
    ),
  );
}

const scene = new Scene({
  canvas,
  layers: [waves()],
});

loop((t) => scene.render(t));
```
