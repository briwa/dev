```sandbox=js viz 480x480
const {
  Scene,
  tween,
  layer,
  line,
  easeInOutSine,
  linear,
  move,
  parallel,
  repeat,
  sequence,
  wait,
} = Canvas;
const COLOR = { r: 224, g: 122, b: 95 };
const BASE_Y = height / 2;
const MAX_RIPPLE = 5;

function drawRipple(y, rippleCount) {
  return line({
    x0: 0,
    x1: width,
    y0: y,
    y1: y,
    color: COLOR,
    segments: 96,
    cycles: 2,
    amplitude: 10,
    phase: 0,
    offset: (t, { amplitude, phase, cycles }) => {
      return amplitude * Math.sin((Math.PI * 2 * t * cycles) + phase);
    },
  });
}

function waves() {
  const group = [];

  const distance = height / 2;
  for (let i = 0; i < MAX_RIPPLE; i++) {
    const phase = i / MAX_RIPPLE;
    const y = BASE_Y + (phase * distance);
    group.push(drawRipple(y, Math.abs(i - MAX_RIPPLE)));
  }
  
  return layer(
    group,
    repeat(
      parallel(
        tween({ phase: 10 }, { duration: 6500, ease: linear }),
        sequence(
          move({ y: -100 }, { duration: 5000, ease: easeInOutSine }),
          wait(1500),
          move({ y: 100 }, { duration: 5000, ease: easeInOutSine }),
        ),
      ),
    ),
  );
};

const scene = new Scene({
  canvas,
  loop: true,
  layers: [waves()],
});

scene.onFinish(() => reset());
loop((t) => scene.render(t));
```
