Back in 2012 or so, I stumbled upon [this article from Mike Bostock](https://web.archive.org/web/20190603060230/https://bost.ocks.org/mike/shuffle/) while working on a card game. As a tribute to the article, I'll walk you through how I attempted to implement the shuffle myself.

At first, I did the exact mistake mentioned in the article. Consider this code below:

```sandbox=js open
function inefficientlyShuffleCards() {
  const cards = getCards();
  const shuffled = [];
  const picked = [];

  while (shuffled.length < cards.length) {
    const idx = Math.floor(Math.random() * cards.length);
    picked.push(idx)
    if (!shuffled.includes(idx)) shuffled.push(idx);
  }

  return { shuffled, picked };
}

```

The very naive logic was indeed to have a list of cards, then take one randomly and put it in a shuffled list, if it's not already there. Here's how the code looks like, visualized:

```sandbox=js viz 400x400 code
function getSteps() {
  const { shuffled, picked } = inefficientlyShuffleCards()
  const steps = [Canvas.move({ y: -Y_SHIFT }, { duration: HOLD / 2 })];

  for (const pick of picked) {
    steps.push(highlight(lineCards[pick]));

    if (shuffled[0] === pick) {
      const x = rowX(Math.abs(shuffled.length - lineCards.length));

      steps.push(Canvas.parallel(
        Canvas.moveTo([lineCards[pick]], { x }, { duration: SHUFFLE }),
        Canvas.move([lineCards[pick]], { y: Y_SHIFT }, { duration: SHUFFLE }),
      ));

      shuffled.shift();
    }
  }

  return steps;
}

render(getSteps());

```

If we inspect the `shuffled` and `picked` output above, `shuffled` would be exactly the size of the original `cards`, but `picked`, if we're not that lucky, would always be bigger than `cards`. There's a _huge chance that we're picking the same shuffled card over and over. The card would _still_ be shuffled eventually, by chance.

At this point, I already gave up on the card game, due to how inconsistent it took to shuffle the cards without knowing why. But the article suggests an obvious fix: shrink the `cards` after picking a random card.

```sandbox=js
function splicedShuffle() {
  const cards = getCards();
  const shuffled = [];

  while (cards.length) {
    const pick = Math.floor(Math.random() * cards.length);
    shuffled.push(cards[pick]);
    cards.splice(pick, 1);
  }

  return shuffled;
}

```

```sandbox=js viz 400x400 code
function getSteps() {
  const shuffled = splicedShuffle();
  const steps = [Canvas.move({ y: -Y_SHIFT }, { duration: HOLD / 2 })];
  const picked = new Set();

  shuffled.forEach((pick, settled) => {
    picked.add(pick);
    const remaining = lineCards.filter((_, idx) => !picked.has(idx));

    steps.push(
      highlight(lineCards[pick]),
      Canvas.parallel(
        Canvas.moveTo([lineCards[pick]], { x: rowX(settled) }, { duration: SHUFFLE }),
        Canvas.move([lineCards[pick]], { y: Y_SHIFT }, { duration: SHUFFLE }),
        Canvas.moveTo(
          remaining,
          (l, i) => ({ x: rowX(i, remaining.length) }),
          { duration: SHUFFLE },
        ),
      ),
    );
  });

  return steps;
}

render(getSteps());

```

Makes sense. But there's an even better way: in-place shuffle, a.k.a [the Fischer-Yates shuffle](https://web.archive.org/web/20190604093712/https://en.wikipedia.org/wiki/Fisher%E2%80%93Yates_shuffle).

```sandbox=js
function fischerYatesShuffle() {
  const cards = getCards();
  const swaps = [];
  let cursor = cards.length;

  while (cursor) {
    const pick = Math.floor(Math.random() * cursor--);
    swap(cards, pick, cursor);
    swaps.push([pick, cursor]);
  }

  return swaps;
}

function swap(cards, before, after) {
  [cards[before], cards[after]] = [cards[after], cards[before]];
}

```

```sandbox=js viz 480x480 code
function getSteps() {
  const swaps = fischerYatesShuffle();
  const steps = [];
  const cards = getCards();

  swaps.forEach(([pick, cursor]) => {
    const picked = cards[pick];
    const settled = cards[cursor];

    steps.push(highlight(lineCards[picked]));

    if (picked !== settled) {
      steps.push(
        Canvas.parallel(
          Canvas.moveTo([lineCards[picked]], { x: rowX(cursor) }, { duration: SHUFFLE }),
          Canvas.moveTo([lineCards[settled]], { x: rowX(pick) }, { duration: SHUFFLE }),
        ),
      );
    }

    swap(cards, pick, cursor);
  });

  return steps;
}

render(getSteps());

```

Instead of shrinking the cards, swap the last unshuffled card with the one we just picked. The next shuffle would ignore that last one, so in a way, the list of cards still shrinks all the same. But, more efficient.

```sandbox=js label="shared helper"
const CARDS_COUNT = 60;
const CARD_HEIGHT = 50;
const CARD_WIDTH = 2;
const SPACING = 6;
const Y_SHIFT = 100;
const DEFAULT_COLOR = { r: 224, g: 122, b: 95 };
const TO_COLOR = { r: 255, g: 0, b: 0 };
const DECK_WIDTH = (CARDS_COUNT - 1) * SPACING + CARD_WIDTH;
const START_X = (width - DECK_WIDTH) / 2;
const START_Y = (height - CARD_HEIGHT) / 2;
const SHUFFLE = 100;
const HOLD = 800;

function getCards() {
  return Array.from({ length: CARDS_COUNT }, (_, idx) => idx);
}

function rowX(i, count = CARDS_COUNT) {
  return (width - ((count - 1) * SPACING + CARD_WIDTH)) / 2 + i * SPACING;
}

function highlight(lineCard) {
  return Canvas.sequence(
    Canvas.tween([lineCard], { color: TO_COLOR }, { duration: SHUFFLE }),
    Canvas.tween([lineCard], { color: DEFAULT_COLOR }, { duration: SHUFFLE }),
  );
}

function render(shuffleSteps) {
  const layers = Canvas.layer(
    lineCards,
    Canvas.sequence(
      ...shuffleSteps,
      Canvas.wait(HOLD),
    ),
  );

  const scene = new Canvas.Scene({ canvas, layers });

  loop((t) => scene.render(t));
  scene.onFinish(() => reset());
}

const lineCards = getCards().map((idx) => {
  const xOffset = idx - CARDS_COUNT / 2;
  const height = Canvas.easeInOutSine(1 - Math.abs(xOffset / CARDS_COUNT)) * CARD_HEIGHT;

  const x0 = rowX(idx);
  const y0 = START_Y + CARD_HEIGHT - height;

  return Canvas.line({
    x0,
    x1: x0 - xOffset,
    y0,
    y1: y0 + height,
    color: DEFAULT_COLOR,
    lineWidth: CARD_WIDTH,
  });
});

```
