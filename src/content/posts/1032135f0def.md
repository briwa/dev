Back in 2012 or so, I had an idea for a turn-based card game, but I got stuck trying to figure out how to shuffle the cards in JavaScript. After googling around, I stumbled upon [this article from Mike Bostock](https://web.archive.org/web/20190603060230/https://bost.ocks.org/mike/shuffle/). I didn't know explaining how shuffling works could be so entertaining. The article became one of my biggest inspirations in programming, and it was probably how I got into data visualization in the first place.

This is my attempt to pay tribute to that article by explaining what happened back then and replicating what it did.

---

I was pretty new to JavaScript at the time. In my head, the logic felt straightforward: if I had a deck of cards, shuffling it meant pulling a random card from it and placing it into a new set.

```js
const cards = [...];
const shuffledCards = [];

while (shuffledCards.length !== cards.length) {
  const randomCardIdx = Math.floor(Math.random() * cards.length);
  shuffledCards.push(cards[randomCardIdx]);
}
```

Wait... I'd be adding duplicate cards that way. So instead, check if it's already there:

```js
if (!hasCard(shuffledCards, randomCardIdx)) {
  shuffledCards.push(cards[randomCardIdx]);
}
```

The code, visualized:

```sandbox=js viz 480x480 code
function inefficientlyShuffleCards() {
  const cards = getCards();
  const shuffled = [];
  const picked = [];

  while (shuffled.length < cards.length) {
    const idx = Math.floor(Math.random() * cards.length);
    picked.push(idx)
    if (!shuffled.includes(idx)) shuffled.push(idx);
  }

  return { shuffled, picked };
}

function getSteps() {
  const { shuffled, picked } = inefficientlyShuffleCards()
  const steps = [Canvas.move({ y: -Y_SHIFT }, { duration: HOLD / 2 })];

  for (const pick of picked) {
    steps.push(highlight(lineCards[pick]));

    if (shuffled[0] === pick) {
      const x = rowX(Math.abs(shuffled.length - lineCards.length));

      steps.push(Canvas.parallel(
        Canvas.moveTo([lineCards[pick]], { x }, { duration: SHUFFLE }),
        Canvas.move([lineCards[pick]], { y: Y_SHIFT }, { duration: SHUFFLE }),
      ));

      shuffled.shift();
    }
  }

  return steps;
}

render(getSteps());

```

Not only does it shuffle the cards randomly, the shuffle also takes a random amount of time (lol). As seen above, it kept picking the already-shuffled cards. To avoid that, it needs to shrink the unshuffled pool after every draw, which is something I was completely oblivious to at the time.

```js
const cards = [...];
const shuffledCards = [];

let i = cards.length;
while (i) {
  const randomCardIdx = Math.floor(Math.random() * i--);
  shuffledCards.push(cards.splice(randomCardIdx, 1));
}
```


```sandbox=js viz 480x480 code
function splicedShuffle() {
  const cards = getCards();
  const shuffled = [];

  while (cards.length) {
    const pick = Math.floor(Math.random() * cards.length);
    shuffled.push(cards[pick]);
    cards.splice(pick, 1);
  }

  return shuffled;
}

function getSteps() {
  const shuffled = splicedShuffle();
  const steps = [Canvas.move({ y: -Y_SHIFT }, { duration: HOLD / 2 })];
  const picked = new Set();

  shuffled.forEach((pick, settled) => {
    picked.add(pick);
    const remaining = lineCards.filter((_, idx) => !picked.has(idx));

    steps.push(
      highlight(lineCards[pick]),
      Canvas.parallel(
        Canvas.moveTo([lineCards[pick]], { x: rowX(settled) }, { duration: SHUFFLE }),
        Canvas.move([lineCards[pick]], { y: Y_SHIFT }, { duration: SHUFFLE }),
        Canvas.moveTo(
          remaining,
          (l, i) => ({ x: rowX(i, remaining.length) }),
          { duration: SHUFFLE },
        ),
      ),
    );
  });

  return steps;
}

render(getSteps());

```


This felt as good as it could get: no re-picking shuffled cards and the shuffle completed in linear time. As it turns out, though, according to the article, there's an even more efficient approach: the Fisher-Yates shuffle. Instead of splicing, the chosen random card is swapped with the last unshuffled card in the deck. The pool of unshuffled cards would still "shrink" the same way, but no splicing involved. In short, an in-place shuffle.


```sandbox=js viz 480x480 code
function fischerYatesShuffle() {
  const cards = getCards();
  const swaps = [];
  let cursor = cards.length;

  while (cursor) {
    const pick = Math.floor(Math.random() * cursor--);
    swap(cards, pick, cursor);
    swaps.push([pick, cursor]);
  }

  return swaps;
}

function swap(cards, before, after) {
  [cards[before], cards[after]] = [cards[after], cards[before]];
}

function getSteps() {
  const swaps = fischerYatesShuffle();
  const steps = [];
  const cards = getCards();

  swaps.forEach(([pick, cursor]) => {
    const picked = cards[pick];
    const settled = cards[cursor];

    steps.push(highlight(lineCards[picked]));

    if (picked !== settled) {
      steps.push(
        Canvas.parallel(
          Canvas.moveTo([lineCards[picked]], { x: rowX(cursor) }, { duration: SHUFFLE }),
          Canvas.moveTo([lineCards[settled]], { x: rowX(pick) }, { duration: SHUFFLE }),
        ),
      );
    }

    swap(cards, pick, cursor);
  });

  return steps;
}

render(getSteps());

```


It's kind of like a physical deck of cards. I could pick a random card and put it at the top of the deck, or putting it next to it, making a new deck. The former is what I'd do usually when I shuffle cards with both my hands. I didn't realize the parallel from the code is pretty close to real life.

Though, despite how much the article blew me away, I never got around to actually build the game.

```sandbox=js label="shared helper"
const CARDS_COUNT = 60;
const CARD_HEIGHT = 50;
const CARD_WIDTH = 2;
const SPACING = 6;
const Y_SHIFT = 100;
const DEFAULT_COLOR = { r: 224, g: 122, b: 95 };
const TO_COLOR = { r: 255, g: 0, b: 0 };
const DECK_WIDTH = (CARDS_COUNT - 1) * SPACING + CARD_WIDTH;
const START_X = (width - DECK_WIDTH) / 2;
const START_Y = (height - CARD_HEIGHT) / 2;
const SHUFFLE = 100;
const HOLD = 800;

function getCards() {
  return Array.from({ length: CARDS_COUNT }, (_, idx) => idx);
}

function rowX(i, count = CARDS_COUNT) {
  return (width - ((count - 1) * SPACING + CARD_WIDTH)) / 2 + i * SPACING;
}

function highlight(lineCard) {
  return Canvas.sequence(
    Canvas.tween([lineCard], { color: TO_COLOR }, { duration: SHUFFLE }),
    Canvas.tween([lineCard], { color: DEFAULT_COLOR }, { duration: SHUFFLE }),
  );
}

function render(shuffleSteps) {
  const layers = Canvas.layer(
    lineCards,
    Canvas.sequence(
      ...shuffleSteps,
      Canvas.wait(HOLD),
    ),
  );

  const scene = new Canvas.Scene({ canvas, layers });

  loop((t) => scene.render(t));
  scene.onFinish(() => reset());
}

const lineCards = getCards().map((idx) => {
  const xOffset = idx - CARDS_COUNT / 2;
  const height = Canvas.easeInOutSine(1 - Math.abs(xOffset / CARDS_COUNT)) * CARD_HEIGHT;

  const x0 = rowX(idx);
  const y0 = START_Y + CARD_HEIGHT - height;

  return Canvas.line({
    x0,
    x1: x0 - xOffset,
    y0,
    y1: y0 + height,
    color: DEFAULT_COLOR,
    lineWidth: CARD_WIDTH,
  });
});

```
